@echo off
REM ─────────────────────────────────────────────────────────────
REM  安裝相依套件（首次使用時跑一次即可）
REM ─────────────────────────────────────────────────────────────

cd /d "%~dp0"

echo.
echo === 升級 pip ===
py -m pip install --upgrade pip

echo.
echo === 安裝套件（這需要 1~3 分鐘）===
py -m pip install -r requirements.txt

echo.
echo === 完成！===
echo 接下來請雙擊 start.bat 啟動程式
echo.
pause
